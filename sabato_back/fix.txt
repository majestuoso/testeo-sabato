export const obtenerTodosForosDB = async () => {$
  return db.any(`$
     SELECT $
  f.foro_id,$
  f.titulo,$
  f.descripcion,$
  f.creador_id,$
  f.fecha_creacion,$
  u.nombre AS "creador_nombre"$
FROM foro f$
LEFT JOIN usuario u ON f.creador_id = u.usuario_id$
ORDER BY f.fecha_creacion DESC$
M-BM-  `);$
};$    arreglo de cierre de order by para que muestre foro
